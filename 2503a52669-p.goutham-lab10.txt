# Find squares of numbers

squares = []
for n in nums:
    squares.append(n**2)
print(len(squares))

OUTPUT:
999999

OPTIMIZED CODE:
# Find squares of numbers using a list comprehension
squares = [n**2 for n in range(1, 1000000)]
print(len(squares))
